{{- /* Image responsive. Paramètres : src ressource image (page bundle ou assets), obligatoire alt texte alternatif ; obligatoire sauf decorative=true sizes attribut sizes (défaut 100vw) widths largeurs candidates (défaut 480 720 1080 1440 1920), plafonnées à la largeur source eager true pour l'image principale au-dessus de la ligne de flottaison class Les GIF et SVG passent tels quels. */ -}} {{- $img := .src -}} {{- $class := .class -}} {{- /* En prévisualisation, validate.html a déjà listé l'erreur : on la marque sur la page sans arrêter le build. */ -}} {{- if not $img }}{{ if not (partial "is-preview.html" .) }}{{ errorf "image.html : paramètre src manquant" }}{{ end }}{{ end -}} {{- if and $img (not .decorative) (not .alt) -}} {{- $_ := partial "content-error.html" (dict "file" "image.html" "where" (printf "photo %s" $img.Name) "msg" "alt manquant, la photo n'a pas de description" "fix" "Ajoutez une description de la photo dans le CMS.") -}} {{- $class = trim (printf "%s erreur-contenu" ($class | default "")) " " -}} {{- end -}} {{- $alt := cond (.decorative | default false) "" (.alt | default "") -}} {{- $loading := cond (.eager | default false) "eager" "lazy" -}} {{- if not $img -}} {{ T "contentErrors.missingImage" }} {{- else if in (slice "svg" "gif") $img.MediaType.SubType -}} {{- else -}} {{- $widths := .widths | default (slice 480 720 1080 1440 1920) -}} {{- $set := slice -}} {{- $largest := false -}} {{- range $widths -}} {{- if le . $img.Width -}} {{- $r := $img.Process (printf "resize %dx webp" .) -}} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink .) -}} {{- $largest = $r -}} {{- end -}} {{- end -}} {{- if not $largest -}} {{- $largest = $img.Process "webp" -}} {{- $set = slice (printf "%s %dw" $largest.RelPermalink $largest.Width) -}} {{- end -}} {{- end -}}
