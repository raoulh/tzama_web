{{- /* Texte éditorial de l'accueil : échappé, espaces insécables de la typographie française, *mots* rendus en surligné au marqueur. Usage : partial "home/text.html" "…". */ -}} {{- $s := . | htmlEscape -}} {{- $s = replaceRE `\s+([:;!?»])` " $1" $s -}} {{- $s = replaceRE `«\s+` "« " $s -}} {{- $s = replaceRE `(\d) (ans|€)` "$1 $2" $s -}} {{- $s = replaceRE `\*([^*]+)\*` `$1` $s -}} {{- return (safeHTML $s) -}}
